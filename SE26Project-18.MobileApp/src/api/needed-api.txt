// ==================== 用户 ====================
getUserById(id)           // 获取用户信息（用于个人主页、头像昵称显示）
updateUser(id, data)      // 更新用户信息
login(username, pwd)      // 登录
register(username, pwd)   // 注册

// ==================== 游戏 ====================
getGames(query='')        // 获取游戏列表（空则为全部）
getGameById(id)           // 获取单个游戏详情

// ==================== 标签 ====================
getGameTags()             // 获取所有游戏标签
getRecruitmentTags()      // 获取所有招募标签

// ==================== 招募 ====================
getRecruitments(gameName='', gameTags[]=[], recruitmentTags[]=[])   // 获取（筛选）招募中的帖子
getRecruitmentsByPublisherId(userId)                                // 获取某用户发布的帖子（含已关闭）
getRecruitmentByChatId(chatId)                                      // 获取某私聊对应的招募（无招募返回id为0的空dto，已删除返回状态为已删除的空dto）
getRecruitmentById(id)                                              // 获取帖子详情（含标签、发布者信息）
saveRecruitment(id=0, data)                                         // 保存帖子（id为0时新建）
deleteRecruitment(id)                                               // 删除帖子

// ==================== 回应 ====================
getResponses(recruitmentId)             // 获取某帖子的所有回应
createResponse(recruitmentId, userId)   // 创建招募回应
deleteResponse(id)                      // 拒绝并删除回应

// ==================== 聊天 ==================== 
getChats(userId)                        // 获取用户的聊天列表（包括已关闭）
getChatById(id)                         // 通过id获取聊天
getChatByUsers(usersId[])               // 通过用户的id获取对应的聊天
createChat(usersId[], recruitmentId=0)  // 创建聊天或刷新招募id
closeChat(id)                           // 举报后关闭聊天（后端通过users_sent_message[]检查并更新限制状态为开放）

// ==================== 消息 ====================
getMessagesByChatId(chatId)             // 获取某私聊的所有消息
sendMessage(chatId, senderId, content)  // 发送消息（包括打招呼消息）
